11void butterworth(
12 unsigned int order,

14 const CoeffType& sampling,

16 std::vector<CoeffType>& denominator
17)
18{
19 // check that the cutoff frequency is less than Nyquist's one
20 if(cutoff*2 > sampling)
21 throw std::runtime_error("In fuction butterworth: cutoff frequency should be less than half of sampling frequency");
22
23 // initialize num/den
24 numerator = std::vector<CoeffType>(1,1);
25 denominator = std::vector<CoeffType>(1,1);
26
27 // If the order is one, this is not really a filter...
28 if(order == 0)
29 return;
30
31 // Evaluate recurring constants
32 const CoeffType gc = std::tan(M_PI*cutoff/sampling);
33 const CoeffType gc2 = gc*gc;
34
35 std::vector<CoeffType> di(3, 1);
36 std::vector<CoeffType> ni(2, 1);
37
38 // evaluate the numerator, which is simply (1+z^-1)^N
39 for(unsigned int i=0; i<order; i++)
40 numerator = polyProd(numerator, ni);
41
42 // evaluate complex conjugate poles
43 for(unsigned int i=0; i<(order/2); i++) {
44 CoeffType ci = 2*gc * std::cos( (order+1+2*i)*M_PI/(2*order) );
45 di[1] = 2 * (gc2-1) / (1+gc2-ci);
46 di[2] = (1+gc2+ci) / (1+gc2-ci);
47 denominator = polyProd(denominator, di);
48 }
49
50 // if 'order' is odd, add a single real pole
51 if(order % 2 > 0)
52 denominator = polyProd(denominator, std::vector<CoeffType>{1, (gc-1)/(gc+1)});
53
54 // normalize
55 CoeffType Sn=numerator[0], Sd=denominator[0];
56 for(unsigned int i=1; i<numerator.size(); i++)
57 Sn = Sn + numerator[i];
58 for(unsigned int i=1; i<denominator.size(); i++)
59 Sd = Sd + denominator[i];
60 for(auto& n : numerator)
61 n = n * Sd / Sn;
62}
