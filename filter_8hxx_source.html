119const DataType& Filter<DataType,CoeffType>::filter(
120 const DataType& x
121)
122{
123 // remove unnecessary input/output
124 in_.pop_back();
125 out_.pop_back();
126
127 // add new input/output
128 in_.push_front(x);
129 out_.push_front(x * b_[0]);
130
131 // complete the filtering
132 auto& y = out_[0];
133
134 for(unsigned int i=1; i<b_.size(); i++)
135 y = y + b_[i] * in_[i];
136 for(unsigned int i=1; i<a_.size(); i++)
137 y = y - a_[i] * out_[i];

143std::vector<DataType> Filter<DataType,CoeffType>::filter(
144 const std::vector<DataType>& x,
145 const std::vector<DataType>& x0,
146 const std::vector<DataType>& y0
147) const
148{
149 // check sizes
150 if(x0.size() != b_.size() - 1) {
151 throw std::runtime_error(
152 "Filter::filter: initial conditions 'x0' have " +
153 std::to_string(x0.size()) + " elements, but " +
154 std::to_string(b_.size()) + " are required"
155 );
156 }
157 if(y0.size() != a_.size() - 1) {
158 throw std::runtime_error(
159 "Filter::filter: initial conditions 'y0' have " +
160 std::to_string(y0.size()) + " elements, but " +
161 std::to_string(a_.size()) + " are required"
162 );
163 }
164
165 // Prepare the output vector
166 std::vector<DataType> y;
167 y.reserve(x.size());
168
169 // main loop: do the filtering
170 for(unsigned int k=0; k<x.size(); k++) {
171 // Init the "current" output from the corresponding input
172 DataType yk = b_[0] * x[k];
173
174 // Add the contributions from past inputs
175 for(unsigned int i=1; i<b_.size(); i++) {
176 if(i > k) {
177 // use values from the initial conditions
178 yk = yk + b_[i] * x0[i-k-1];
179 }
180 else {
181 // use values from the input vector
182 yk = yk + b_[i] * x[k-i];
183 }
184 }
185
186 // Add the contributions from past outputs
187 for(unsigned int i=1; i<a_.size(); i++) {
188 if(i > k) {
189 // use values from the initial conditions
190 yk = yk - a_[i] * y0[i-k-1];
191 }
192 else {
193 // use values from the input vector
194 yk = yk - a_[i] * y[k-i];
195 }
196 }
197
198 // push the computed value into the result vector
199 y.push_back(yk);
200 }
201
202 // return the filtered vector
203 return y;
204}
